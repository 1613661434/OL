339 template <typename LockType = spin_mutex>
340 class clogfile // class log file
341 {
342 private:
343 std::ofstream fout;
344 std::string m_filename;
345 std::ios::openmode m_mode;
346 LockType m_lock;
347 bool m_enBuffer;
348 bool m_isRoll;
349 long m_maxSize;
350
351 public:
356 clogfile() : m_mode(std::ios::app), m_enBuffer(false), m_isRoll(true), m_maxSize(100) {}
357
358 // 析构函数，自动关闭文件
359 ~clogfile() { close(); };
360
361 // 关闭日志文件
362 void close()
363 {
364 std::lock_guard<LockType> lock(m_lock);
365 if (fout.is_open())
366 {
367 fout.flush();
368 fout.close();
369 }
370 }
371
381 bool open(const std::string& filename, std::ios::openmode mode = std::ios::app, bool isRoll = true, long maxSize = 100, bool enBuffer = false)
382 {
383 if (isRoll && maxSize == 0)
384 {
385 fprintf(stderr, "Error: When isRoll is true, maxSize must be greater than 0\n");
386 return false;
387 }
388
389 std::lock_guard<LockType> lock(m_lock);
390
391 // 如果日志文件是打开的状态，先关闭它。
392 if (fout.is_open())
393 {
394 fout.flush();
395 fout.close();
396 }
397
398 m_filename = filename; // 日志文件名。
399 m_mode = mode | std::ios::out; // 打开模式。
400 m_isRoll = isRoll; // 是否自动备份。
401 m_maxSize = maxSize; // 备份文件最大容量（MB）
402 m_enBuffer = enBuffer; // 是否启用文件缓冲区。
403
404 newdir(m_filename, true); // 如果日志文件的目录不存在，创建它。
405
406 fout.open(m_filename, m_mode); // 打开日志文件。
407
408 if (!m_enBuffer) fout << std::unitbuf; // 是否启用文件缓冲区。
409
410 return fout.is_open();
411 }
412

422 {
423 std::lock_guard<LockType> lock(m_lock);
424
425 if (!fout.is_open()) return false;

427 if (m_isRoll && !rolling()) fout << ltime1() << ' ' << "[ERROR]:Rolling failed\n";
428

529 template <typename T>
530 clogfile& operator<<(const T& value)
531 {
532 std::lock_guard<LockType> lock(m_lock);
533
534 if (!fout.is_open()) return *this;
535
536 if (m_isRoll && !rolling()) fout << ltime1() << ' ' << "[ERROR]:Rolling failed\n";
537
538 fout << value;
539
540 return *this;
541 }
542
543 private:
549 bool rolling()
550 {
551 if (!fout.is_open()) return false;
552
553 fout.flush();
554
555 long fsize = filesize(m_filename);
556 if (fsize == -1) return false;
557
558 // 如果当前日志文件的大小超过m_maxSize，备份日志。
559 if (fsize >= m_maxSize * 1024 * 1024)
560 {
561 fout.close(); // 关闭当前日志文件。
562
563 // 拼接备份日志文件名。
564 size_t log_suffix_pos = m_filename.rfind(".log");
565 std::string bak_filename = m_filename.substr(0, log_suffix_pos) + ltime1("yyyymmddhh24miss") + ".log";
566
567 // 把当前日志文件改名为备份日志文件。
568 if (!renamefile(m_filename, bak_filename))
569 {
570 fout.open(m_filename, m_mode); // 重命名失败，重新打开原文件
571 if (!m_enBuffer) fout << std::unitbuf;
572 return false;
573 }
574
575 fout.open(m_filename, m_mode); // 重新打开当前日志文件。
576
577 if (!m_enBuffer) fout << std::unitbuf; // 判断是否启动文件缓冲区。
578
579 return fout.is_open();
580 }
581
582 return true;
583 }
