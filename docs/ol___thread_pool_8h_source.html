210 template <bool D = IsDynamic, typename = std::enable_if_t<!D>>
211 ThreadPool(size_t threadNum, size_t maxQueueSize = 0)
212 : m_state(State::Running), m_activeWorkers(0),
213 m_maxQueueSize(maxQueueSize),
214 m_queueFullPolicy(QueueFullPolicy::kReject), m_timeoutMS(std::chrono::milliseconds(500))

227#endif

248 template <bool D = IsDynamic, typename = std::enable_if_t<D>>
249 ThreadPool(size_t minThreadNum = 0,
250 size_t maxThreadNum = std::thread::hardware_concurrency(),
251 size_t maxQueueSize = 0,
252 std::chrono::seconds checkInterval = std::chrono::seconds(1))
253 : m_state(State::Running), m_activeWorkers(0),
254 m_maxQueueSize(maxQueueSize),
255 m_queueFullPolicy(QueueFullPolicy::kReject), m_timeoutMS(std::chrono::milliseconds(500))
256 {
257 if (minThreadNum > maxThreadNum) throw std::invalid_argument("[ol::ThreadPool] Invalid thread number range");
258
259 if (minThreadNum == maxThreadNum && minThreadNum == 0)
260 {
261 m_state.store(State::Stopped, std::memory_order_release);
262#ifdef OL_DEBUG
263 debugLog("pool-create", "state=Stopped min_workers=0 max_workers=0 max_queue=%zu", m_maxQueueSize);
264#endif
265 return;
266 }
267
268 // 最小线程数至少为1
269 minThreadNum = std::max(minThreadNum, static_cast<size_t>(1));
270
271 // 初始化动态模式成员
272 m_dynamic.minThreads = minThreadNum;
273 m_dynamic.maxThreads = maxThreadNum;
274 m_dynamic.idleThreads = 0;
275 m_dynamic.workerExitNum = 0;
276 m_dynamic.checkInterval = checkInterval;
277
278#ifdef OL_DEBUG
279 debugLog("pool-create",
280 "state=Running min_workers=%zu max_workers=%zu max_queue=%zu check_interval_s=%lld",
281 m_dynamic.minThreads,
282 m_dynamic.maxThreads,
283 m_maxQueueSize,
284 static_cast<long long>(m_dynamic.checkInterval.count()));
285#endif
286
287 m_workers.reserve(minThreadNum);
288 while (minThreadNum > 0)
289 {
290 m_activeWorkers.fetch_add(1, std::memory_order_acq_rel);
291 std::thread th(&ThreadPool<IsDynamic>::worker, this);
292 m_workers.emplace(th.get_id(), std::move(th)); // 移动到哈希表
293 --minThreadNum;
294 }
295
296 // 启动管理者线程
297 m_dynamic.managerThread = std::thread(&ThreadPool<IsDynamic>::manager<IsDynamic>, this);
298 }

324 void requestStop(ShutdownMode mode = ShutdownMode::Drain)
325 {
326 std::queue<std::function<void()>> cancelledTasks;
327 bool stateChanged = false;
328 State previousState;
329 State nextState;
330#ifdef OL_DEBUG
331 size_t queuedBefore = 0;
332 size_t droppedTasks = 0;
333#endif
334 {
335 std::lock_guard<std::mutex> lock(m_taskQueueMutex);
336 previousState = m_state.load(std::memory_order_acquire);
337 nextState = previousState;
338#ifdef OL_DEBUG
339 queuedBefore = m_taskQueue.size();
340#endif
341
342 if (mode == ShutdownMode::CancelPending)
343 {
344 if (previousState == State::Running || previousState == State::Draining)
345 {
346 nextState = State::Cancelling;
347 m_state.store(nextState, std::memory_order_release);
348#ifdef OL_DEBUG
349 droppedTasks = m_taskQueue.size();
350#endif
351 cancelledTasks.swap(m_taskQueue);
352 stateChanged = true;
353 }
354 }
355 else if (previousState == State::Running)
356 {
357 nextState = State::Draining;
358 m_state.store(nextState, std::memory_order_release);
359 stateChanged = true;
360 }
361
362#ifdef OL_DEBUG
363 if (stateChanged)
364 {
365 debugLog("state-change",
366 "from=%s to=%s queued_before=%zu dropped=%zu active_workers=%zu",
367 stateName(previousState),
368 stateName(nextState),
369 queuedBefore,
370 droppedTasks,
371 m_activeWorkers.load(std::memory_order_acquire));
372 }
373#endif
374 }
375
376 if (!stateChanged) return;
377
378 if constexpr (IsDynamic) m_dynamic.managerExit_condVar.notify_all();
379 m_taskQueueNotEmpty_condVar.notify_all();
380 m_taskQueueNotFull_condVar.notify_all();
381 }

388 void wait()
389 {
390 if (s_currentThreadPool == this)
391 {
392#ifdef OL_DEBUG
393 debugLog("invalid-call", "wait called from an internal thread");
394#endif
395 throw std::logic_error("[ol::ThreadPool] wait() cannot be called from an internal thread");
396 }
397
398 std::unique_lock<std::mutex> waitLock(m_waitMutex);
399 State state = m_state.load(std::memory_order_acquire);
400 if (state == State::Stopped) return;
401 if (state == State::Running)
402 {
403#ifdef OL_DEBUG
404 debugLog("invalid-call", "wait called while state=Running");
405#endif
406 throw std::logic_error("[ol::ThreadPool] requestStop() must be called before wait()");
407 }
408
409#ifdef OL_DEBUG
410 debugLog("wait-begin",
411 "state=%s queued=%zu active_workers=%zu",
412 stateName(state),
413 getTaskNum(),
414 m_activeWorkers.load(std::memory_order_acquire));
415#endif
416
417 if constexpr (IsDynamic)
418 {
419 if (m_dynamic.managerThread.joinable())
420 {
421 m_dynamic.managerThread.join();
422#ifdef OL_DEBUG
423 debugLog("manager-joined", "state=%s", stateName(m_state.load(std::memory_order_acquire)));
424#endif
425 }
426 }
427
428 std::vector<std::thread> workers;
429 {
430 std::lock_guard<std::mutex> lock(m_workersMutex);
431 workers.reserve(m_workers.size());
432 if constexpr (IsDynamic)
433 {
434 for (auto& [id, workerThread] : m_workers)
435 {
436 (void)id;
437 workers.push_back(std::move(workerThread));
438 }
439 m_workers.clear();
440 }
441 else
442 {
443 workers.swap(m_workers);
444 }
445 }
446
447#ifdef OL_DEBUG
448 debugLog("worker-join-begin", "count=%zu", workers.size());
449#endif
450 for (auto& workerThread : workers)
451 {
452 if (workerThread.joinable()) workerThread.join();
453 }
454
455 if constexpr (IsDynamic)
456 {
457 std::lock_guard<std::mutex> lock(m_dynamic.workerExitId_dequeMutex);
458 m_dynamic.workerExitId_deque.clear();
459 }
460
461#ifdef OL_DEBUG
462 const State shutdownState = m_state.load(std::memory_order_acquire);
463#endif
464 m_state.store(State::Stopped, std::memory_order_release);
465#ifdef OL_DEBUG
466 debugLog("wait-end",
467 "from=%s to=Stopped joined=%zu active_workers=%zu",
468 stateName(shutdownState),
469 workers.size(),
470 m_activeWorkers.load(std::memory_order_acquire));
471#endif
472 }

619 bool addTask(std::function<void()> task)
620 {
621 if (m_state.load(std::memory_order_acquire) != State::Running) return false;
622
623 {
624 std::unique_lock<std::mutex> lock(m_taskQueueMutex);
625
626 // 处理队列大小限制
627 if (m_maxQueueSize > 0)
628 {
629 while (m_taskQueue.size() >= m_maxQueueSize &&
630 m_state.load(std::memory_order_acquire) == State::Running)
631 {
632 switch (m_queueFullPolicy)
633 {
634 case QueueFullPolicy::kReject:
635 return false;
636 case QueueFullPolicy::kBlock:
637 m_taskQueueNotFull_condVar.wait(lock, [this]()
638 { return m_taskQueue.size() < m_maxQueueSize || m_state.load(std::memory_order_acquire) != State::Running; });
639 break;
640 case QueueFullPolicy::kTimeout:
641 bool result = m_taskQueueNotFull_condVar.wait_for(lock, m_timeoutMS,
642 [this]()
643 { return m_taskQueue.size() < m_maxQueueSize || m_state.load(std::memory_order_acquire) != State::Running; });
644 if (!result) return false;
645 break;
646 }
647 }
648 }
649
650 if (m_state.load(std::memory_order_acquire) != State::Running) return false;
651 m_taskQueue.push(std::move(task));
652 }
653
654 m_taskQueueNotEmpty_condVar.notify_one();
655 return true;
656 }

669 template <typename F, typename... Args>
670 auto submitTask(F&& f, Args&&... args) -> std::pair<bool, std::future<typename std::invoke_result_t<F, Args...>>>
671 {
672 using ReturnType = typename std::invoke_result_t<F, Args...>;
673
674 auto task = std::make_shared<std::packaged_task<ReturnType()>>(
675 [f = std::forward<F>(f), args = std::make_tuple(std::forward<Args>(args)...)]() mutable
676 {
677 return std::apply(std::move(f), std::move(args));
678 });
679
680 std::future<ReturnType> result = task->get_future();
681
682 if (!addTask([task]() mutable
683 { (*task)(); }))
684 {
685 std::promise<ReturnType> promise;
686 State state;
687 QueueFullPolicy queueFullPolicy;
688 {
689 std::lock_guard<std::mutex> lock(m_taskQueueMutex);
690 state = m_state.load(std::memory_order_acquire);
691 queueFullPolicy = m_queueFullPolicy;
692 }
693
694 if (state != State::Running)
695 {
696 promise.set_exception(std::make_exception_ptr(std::runtime_error("[ol::ThreadPool] ThreadPool is not accepting tasks")));
697 return {false, promise.get_future()};
698 }
699
700 switch (queueFullPolicy)
701 {
702 case QueueFullPolicy::kReject:
703 promise.set_exception(std::make_exception_ptr(std::runtime_error("[ol::ThreadPool] Task queue full (Reject policy)")));
704 return {false, promise.get_future()};
705 case QueueFullPolicy::kBlock:
706 // 此时失败一定是因为线程池已停止（否则wait会一直等）
707 promise.set_exception(std::make_exception_ptr(std::runtime_error("[ol::ThreadPool] Task submission failed in block policy (ThreadPool stopped)")));
708 return {false, promise.get_future()};
709 case QueueFullPolicy::kTimeout:
710 promise.set_exception(std::make_exception_ptr(std::runtime_error("[ol::ThreadPool] Task queue full (Timeout policy)")));
711 return {false, promise.get_future()};
712 default:
713 promise.set_exception(std::make_exception_ptr(std::runtime_error("[ol::ThreadPool] Task submission failed")));
714 return {false, promise.get_future()};
715 }
716 }
717
718 return {true, std::move(result)};
719 }

737 void worker()
738 {
739 ThreadContextGuard threadContext(this);
740#ifdef OL_DEBUG
741 const char* exitReason = "unknown";
742 debugLog("worker-start",
743 "state=%s active_workers=%zu",
744 stateName(m_state.load(std::memory_order_acquire)),
745 m_activeWorkers.load(std::memory_order_acquire));
746#endif
747
748 // 动态模式：初始化线程状态
749 if constexpr (IsDynamic)
750 {
751 m_dynamic.idleThreads.fetch_add(1, std::memory_order_release);
752 }
753
754 try
755 {
756 while (true)
757 {
758 std::function<void()> task;
759
760 {
761 std::unique_lock<std::mutex> lock(m_taskQueueMutex);
762
763 // 等待任务或停止信号
764 auto waitCond = [this]()
765 {
766 if constexpr (IsDynamic)
767 return !m_taskQueue.empty() || m_state.load(std::memory_order_acquire) != State::Running || m_dynamic.workerExitNum.load(std::memory_order_acquire) > 0;
768 else
769 return !m_taskQueue.empty() || m_state.load(std::memory_order_acquire) != State::Running;
770 };
771 m_taskQueueNotEmpty_condVar.wait(lock, waitCond);
772
773 const State state = m_state.load(std::memory_order_acquire);
774 if (state == State::Cancelling)
775 {
776#ifdef OL_DEBUG
777 exitReason = "cancelled";
778#endif
779 break;
780 }
781 if (state == State::Stopped)
782 {
783#ifdef OL_DEBUG
784 exitReason = "stopped";
785#endif
786 break;
787 }
788 if (state == State::Draining && m_taskQueue.empty())
789 {
790#ifdef OL_DEBUG
791 exitReason = "drained";
792#endif
793 break;
794 }
795
796 if constexpr (IsDynamic)
797 {
798 if (state == State::Running &&
799 m_dynamic.workerExitNum.load(std::memory_order_acquire) > 0)
800 {
801 m_dynamic.workerExitNum.fetch_sub(1, std::memory_order_acq_rel);
802#ifdef OL_DEBUG
803 exitReason = "scale-down";
804#endif
805 break;
806 }
807 }
808
809 if (m_taskQueue.empty()) continue;
810
811 // 取出任务
812 task = std::move(m_taskQueue.front());
813 m_taskQueue.pop();
814
815 // 通知可能等待的生产者
816 if (m_queueFullPolicy != QueueFullPolicy::kReject) m_taskQueueNotFull_condVar.notify_one();
817
818 // 动态模式：空闲线程数-1
819 if constexpr (IsDynamic) m_dynamic.idleThreads.fetch_sub(1, std::memory_order_acq_rel);
820 }
821
822 // 执行任务
823 try
824 {
825 if (task) task(); // 空任务保护
826 }
827 catch (const std::exception& e)
828 {
829 errorLog("task-error", "exception=%s", e.what());
830 }
831 catch (...)
832 {
833 errorLog("task-error", "unknown exception");
834 }
835
836 // 动态模式：任务完成，恢复空闲状态
837 if constexpr (IsDynamic)
838 {
839 m_dynamic.idleThreads.fetch_add(1, std::memory_order_release);
840 }
841 }
842 }
843 catch (const std::exception& e)
844 {
845#ifdef OL_DEBUG
846 exitReason = "exception";
847#endif
848 errorLog("worker-error", "exception=%s", e.what());
849 }
850 catch (...)
851 {
852#ifdef OL_DEBUG
853 exitReason = "exception";
854#endif
855 errorLog("worker-error", "unknown exception");
856 }
857
858 // 活跃线程数-1
859#ifdef OL_DEBUG
860 const size_t remainingActiveWorkers = m_activeWorkers.fetch_sub(1, std::memory_order_acq_rel) - 1;
861#else
862 m_activeWorkers.fetch_sub(1, std::memory_order_acq_rel);
863#endif
864
865 // 动态模式：空闲线程数-1
866 if constexpr (IsDynamic)
867 {
868 m_dynamic.idleThreads.fetch_sub(1, std::memory_order_acq_rel);
869 // 线程池未停止时，让管理者清理线程
870 if (m_state.load(std::memory_order_acquire) == State::Running)
871 {
872 std::unique_lock<std::mutex> lock_exitVector(m_dynamic.workerExitId_dequeMutex);
873 m_dynamic.workerExitId_deque.emplace_back(std::this_thread::get_id());
874 }
875 }
876
877#ifdef OL_DEBUG
878 debugLog("worker-exit", "reason=%s active_workers=%zu", exitReason, remainingActiveWorkers);
879#endif
880 }
881
890 template <bool D = IsDynamic, typename = std::enable_if_t<D>>
891 void manager()
892 {
893 ThreadContextGuard threadContext(this);
894
895#ifdef OL_DEBUG
896 debugLog("manager-start",
897 "state=%s min_workers=%zu max_workers=%zu",
898 stateName(m_state.load(std::memory_order_acquire)),
899 m_dynamic.minThreads,
900 m_dynamic.maxThreads);
901#endif
902
903 try
904 {
905 std::deque<std::thread::id> exitIds;
906 while (m_state.load(std::memory_order_acquire) == State::Running)
907 {
908 // 定期检查（可被stop()唤醒）
909 std::unique_lock<std::mutex> lock_manger(m_dynamic.managerMutex);
910 m_dynamic.managerExit_condVar.wait_for(lock_manger, m_dynamic.checkInterval, [this]()
911 { return m_state.load(std::memory_order_acquire) != State::Running; });
912 if (m_state.load(std::memory_order_acquire) != State::Running) break;
913 lock_manger.unlock();
914
915 // 1. 清理已终止的线程对象
916#ifdef OL_DEBUG
917 size_t joinedWorkers = 0;
918 size_t missingWorkers = 0;
919#endif
920 {
921 // 上锁
922 std::lock_guard<std::mutex> lock_workers(m_workersMutex);
923 std::unique_lock<std::mutex> lock_exitDeque(m_dynamic.workerExitId_dequeMutex);
924
925 // 交换退出线程ID队列
926 exitIds.clear();
927 exitIds.swap(m_dynamic.workerExitId_deque);
928 lock_exitDeque.unlock(); // 解锁
929
930 // 清理退出的线程
931 for (const auto& exitId : exitIds)
932 {
933 auto it = m_workers.find(exitId);
934 if (it == m_workers.end())
935 {
936#ifdef OL_DEBUG
937 ++missingWorkers;
938#endif
939 continue;
940 }
941
942 if (it->second.joinable())
943 {
944 try
945 {
946 it->second.join();
947#ifdef OL_DEBUG
948 ++joinedWorkers;
949#endif
950 }
951 catch (const std::exception& e)
952 {
953 errorLog("worker-join-error",
954 "worker_std_id=%" PRIu64 " exception=%s",
955 getThreadId(exitId),
956 e.what());
957 }
958 catch (...)
959 {
960 errorLog("worker-join-error",
961 "worker_std_id=%" PRIu64 " unknown exception",
962 getThreadId(exitId));
963 }
964 }
965
966 m_workers.erase(it);
967 }
968 }
969#ifdef OL_DEBUG
970 if (joinedWorkers > 0 || missingWorkers > 0)
971 {
972 debugLog("worker-cleanup",
973 "joined=%zu missing=%zu remaining_workers=%zu",
974 joinedWorkers,
975 missingWorkers,
976 getWorkerNum());
977 }
978#endif
979
980 // 2. 扩缩容
981 size_t addedThreads = 0;
982 size_t reducedThreads = 0;
983 size_t taskCountSnapshot = 0;
984 size_t workerCountSnapshot = 0;
985 size_t idleCountSnapshot = 0;
986#ifdef OL_DEBUG
987 size_t reducedThreadsForLog = 0;
988 size_t workerCountAfterScale = 0;
989#endif
990 {
991 std::lock_guard<std::mutex> lock_taskQueue(m_taskQueueMutex);
992 std::lock_guard<std::mutex> lock_workers(m_workersMutex);
993 taskCountSnapshot = m_taskQueue.size();
994 workerCountSnapshot = m_workers.size();
995 idleCountSnapshot = m_dynamic.idleThreads.load(std::memory_order_acquire);
996
997 // 扩容判断：任务数 > 线程数 * 2 且未达最大线程数
998 if (taskCountSnapshot > workerCountSnapshot * 2 && workerCountSnapshot < m_dynamic.maxThreads)
999 {
1000 addedThreads = std::min(
1001 m_dynamic.maxThreads - workerCountSnapshot,
1002 (taskCountSnapshot + workerCountSnapshot - 1) / workerCountSnapshot // 按当前负载估算需要的线程数（向上取整）
1003 );
1004 // 每次最多扩容到当前的1.5倍，避免一次性创建过多线程
1005 addedThreads = std::min(addedThreads, workerCountSnapshot / 2 + 1);
1006
1007 m_workers.reserve(workerCountSnapshot + addedThreads);
1008 for (size_t i = 0; i < addedThreads; ++i)
1009 {
1010 m_activeWorkers.fetch_add(1, std::memory_order_acq_rel);
1011 std::thread th(&ThreadPool<IsDynamic>::worker, this);
1012 m_workers.emplace(th.get_id(), std::move(th)); // 哈希表插入新工作线程
1013 }
1014 }
1015 // 缩容判断：空闲线程 > 线程总数的1/2 且 线程数 > 最小线程数
1016 else if (idleCountSnapshot > workerCountSnapshot / 2 && workerCountSnapshot > m_dynamic.minThreads)
1017 {
1018 // 实际缩减数 = 取 可缩减线程数 和 多余空闲线程数 的较小值
1019 reducedThreads = std::min(
1020 workerCountSnapshot - m_dynamic.minThreads, // 可缩减线程数 = 当前线程数 - 最低保留数
1021 idleCountSnapshot - (workerCountSnapshot / 2) // 多余空闲线程数 = 超过一半的空闲线程
1022 );
1023#ifdef OL_DEBUG
1024 reducedThreadsForLog = reducedThreads;
1025#endif
1026
1027 // 销毁线程：设置退出数量后逐个唤醒，避免惊群效应
1028 if (reducedThreads > 0)
1029 {
1030 m_dynamic.workerExitNum.fetch_add(reducedThreads, std::memory_order_acq_rel);
1031 do
1032 {
1033 m_taskQueueNotEmpty_condVar.notify_one();
1034 --reducedThreads;
1035 } while (reducedThreads > 0);
1036 }
1037 }
1038#ifdef OL_DEBUG
1039 workerCountAfterScale = m_workers.size();
1040#endif
1041 }
1042#ifdef OL_DEBUG
1043 if (addedThreads > 0)
1044 {
1045 debugLog("scale-up",
1046 "from=%zu to=%zu added=%zu queued=%zu idle=%zu",
1047 workerCountSnapshot,
1048 workerCountAfterScale,
1049 addedThreads,
1050 taskCountSnapshot,
1051 idleCountSnapshot);
1052 }
1053 else if (reducedThreadsForLog > 0)
1054 {
1055 debugLog("scale-down-request",
1056 "current=%zu requested_exit=%zu queued=%zu idle=%zu min_workers=%zu",
1057 workerCountSnapshot,
1058 reducedThreadsForLog,
1059 taskCountSnapshot,
1060 idleCountSnapshot,
1061 m_dynamic.minThreads);
1062 }
1063#endif
1064 }
1065 }
1066 catch (const std::exception& e)
1067 {
1068 errorLog("manager-error", "exception=%s", e.what());
1069 }
1070 catch (...)
1071 {
1072 errorLog("manager-error", "unknown exception");
1073 }
1074
1075 // 清空退出队列
1076 {
1077 std::lock_guard<std::mutex> lock_exit_deque(m_dynamic.workerExitId_dequeMutex);
1078 m_dynamic.workerExitId_deque.clear();
1079 }
1080#ifdef OL_DEBUG
1081 debugLog("manager-exit", "state=%s", stateName(m_state.load(std::memory_order_acquire)));
1082#endif
1083 }
1084 };
