1/****************************************************************************************/
2/*
3 * 程序名：ol_database.h
4 * 功能描述：数据库连接抽象基类 IDBConn + 模板连接池 DBPool，支持以下特性：
5 * - 线程安全：通过互斥锁和条件变量保证多线程环境下连接获取/释放安全
6 * - 阻塞等待：无空闲连接时线程自动阻塞，不占用CPU资源
7 * - 超时获取：支持带默认超时的连接获取，避免无限阻塞
8 * - 自动重连：获取连接时自动检测有效性，失效则重新连接
9 * - 资源管理：连接池销毁时自动关闭所有连接，无资源泄漏
10 * - 单例模式：基于CRTP单例基类，每种数据库类型对应一个全局单例
11 * 规范：所有数据库 DBConn 必须继承 IDBConn
12 * 作者：ol
13 * 适用标准：C++17及以上
14 */
15/****************************************************************************************/

196 void returnConnection(StoragePtr conn) noexcept
197 {
198 if (!conn) return;
199
200 try
201 {
202 conn->reset();
203 }
204 catch (...)
205 {
206 disconnectNoexcept(conn);
207 return;
208 }
209
210 bool returned = false;
211 {
212 std::lock_guard<std::mutex> lock(m_mtx);
213 if (m_state == State::Running)
214 {
215 try
216 {
217 m_queue.push(std::move(conn));
218 returned = true;
219 }
220 catch (...)
221 {
222 }
223 }
224 }
225
226 if (returned)
227 m_cv.notify_one();
228 else
229 disconnectNoexcept(conn);
230 }

232 ConnPtr prepareLease(StoragePtr conn, const ConnConfigCallback& config_cb)
233 {
234 try
235 {
236 if (!conn->isConnected())

238 config_cb(*conn);

241 returnConnection(std::move(conn));

244 }
245 }

263 void init(size_t max_conn, ConnConfigCallback config_cb)
264 {
265 if (!config_cb) throw std::invalid_argument("DBPool: Connection configuration callback cannot be empty!");
266 if (max_conn == 0) throw std::invalid_argument("DBPool: Maximum connection count cannot be zero!");
267
268 {
269 std::lock_guard<std::mutex> lock(m_mtx);
270 if (m_state != State::Uninitialized)
271 throw std::runtime_error("DBPool: Singleton pool has been initialized or stopped!");
272 m_state = State::Initializing;
273 }
274
275 std::queue<StoragePtr> connections;
276 try
277 {
278 for (size_t i = 0; i < max_conn; ++i)
279 {
280 StoragePtr conn = std::make_unique<T>();
281 config_cb(*conn);
282 if (!conn->connect())
283 {
284 disconnectNoexcept(conn);
285 throw std::runtime_error("DBPool: Database connection initialization failed!");
286 }
287 connections.push(std::move(conn));
288 }
289 }
290 catch (...)
291 {
292 disconnectAll(connections);
293 {
294 std::lock_guard<std::mutex> lock(m_mtx);
295 if (m_state == State::Initializing) m_state = State::Uninitialized;
296 }
297 m_cv.notify_all();
298 throw;
299 }
300
301 bool initialized = false;
302 {
303 std::lock_guard<std::mutex> lock(m_mtx);
304 if (m_state == State::Initializing)
305 {
306 m_max_conn = max_conn;
307 m_config_cb = std::move(config_cb);
308 m_queue.swap(connections);
309 m_state = State::Running;
310 initialized = true;
311 }
312 }
313
314 m_cv.notify_all();
315 if (initialized) return;
316
317 disconnectAll(connections);
318 throw std::runtime_error("DBPool: Pool was stopped during initialization!");
319 }

327 ConnPtr get()
328 {
329 StoragePtr conn;
330 ConnConfigCallback config_cb;
331 std::unique_lock<std::mutex> lock(m_mtx);
332
333 if (m_state == State::Uninitialized)
334 throw std::logic_error("DBPool::get: Pool has not been initialized!");
335
336 if (m_state == State::Initializing)
337 m_cv.wait(lock, [this]() { return m_state != State::Initializing; });
338
339 if (m_state != State::Running) return {};
340
341 m_cv.wait(lock, [this]()
342 { return !m_queue.empty() || m_state != State::Running; });
343
344 if (m_state != State::Running) return {};
345
346 conn = std::move(m_queue.front());
347 m_queue.pop();
348 config_cb = m_config_cb;
349 lock.unlock();
350
351 return prepareLease(std::move(conn), config_cb);
352 }

361 ConnPtr getTimeout(TimeoutMs timeout_ms = TimeoutMs(3000))
362 {
363 if (timeout_ms < TimeoutMs::zero())
364 throw std::invalid_argument("DBPool::getTimeout: Timeout cannot be negative!");
365
366 const auto deadline = std::chrono::steady_clock::now() + timeout_ms;
367 StoragePtr conn;
368 ConnConfigCallback config_cb;
369 std::unique_lock<std::mutex> lock(m_mtx);
370
371 if (m_state == State::Uninitialized)
372 throw std::logic_error("DBPool::getTimeout: Pool has not been initialized!");
373
374 if (m_state == State::Initializing &&
375 !m_cv.wait_until(lock, deadline, [this]() { return m_state != State::Initializing; }))
376 return {};
377
378 if (m_state != State::Running) return {};
379
380 if (!m_cv.wait_until(lock, deadline, [this]()
381 { return !m_queue.empty() || m_state != State::Running; }))
382 return {};
383
384 if (m_state != State::Running) return {};
385
386 conn = std::move(m_queue.front());
387 m_queue.pop();
388 config_cb = m_config_cb;
389 lock.unlock();
390
391 return prepareLease(std::move(conn), config_cb);
392 }
