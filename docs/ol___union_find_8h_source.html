43 template <typename T>
44 class UnionFind<T, std::enable_if_t<std::is_integral_v<T>>>
45 {
46 private:
47 std::vector<T> parent;
48 std::vector<size_t> rank;
49
50 public:
55 explicit UnionFind(size_t size = 0) { init(size); }
56
61 void init(size_t size)
62 {
63 parent.resize(size);
64 rank.resize(size, 0);
65 for (size_t i = 0; i < size; ++i) parent[i] = static_cast<T>(i);
66 }
67
75 T find(T x)
76 {
77 if (parent[x] != x) parent[x] = find(parent[x]); // 递归压缩路径
78 return parent[x];
79 }
80
88 void unite(T x, T y)
89 {
90 T rootX = find(x);
91 T rootY = find(y);
92
93 if (rootX == rootY) return; // 已在同一集合
94
95 if (rank[rootX] < rank[rootY])
96 {
97 parent[rootX] = rootY;
98 }
99 else if (rank[rootX] > rank[rootY])
100 {
101 parent[rootY] = rootX;
102 }
103 else
104 {
105 parent[rootY] = rootX;
106 ++rank[rootX]; // 秩相同，合并后根节点秩+1
107 }
108 }
109
116 bool connected(T x, T y) { return find(x) == find(y); }
117
122 size_t countSets() const
123 {
124 size_t count = 0;
125 for (size_t i = 0; i < parent.size(); ++i)
126 {
127 // 根节点的父节点是自身
128 if (parent[i] == static_cast<T>(i)) ++count;
129 }
130 return count;
131 }
132
137 inline size_t size() const { return parent.size(); }
138 };

146 template <typename T>
147 class UnionFind<T, std::enable_if_t<!std::is_integral_v<T>>>
148 {
149 private:
153 struct Node
154 {
155 T parent;
156 size_t rank;
162 Node(const T& p = T(), size_t r = 0) : parent(p), rank(r) {}
163 };
164
165 private:
166 std::unordered_map<T, Node> nodes;
167
168 public:
173 void insert(const T& x)
174 {
175 if (nodes.find(x) == nodes.end()) nodes[x] = Node(x); // 新元素初始父节点为自身
176 }
177
186 T find(const T& x)
187 {
188 auto it = nodes.find(x);
189 if (it == nodes.end())
190 {
191 insert(x); // 自动插入不存在的元素
192 return x;
193 }
194
195 if (it->second.parent != x) it->second.parent = find(it->second.parent); // 递归压缩路径
196 return it->second.parent;
197 }
198
206 void unite(const T& x, const T& y)
207 {
208 insert(x);
209 insert(y);
210
211 T rootX = find(x);
212 T rootY = find(y);
213
214 if (rootX == rootY) return; // 已在同一集合
215
216 auto& nodeX = nodes[rootX];
217 auto& nodeY = nodes[rootY];
218
219 if (nodeX.rank < nodeY.rank)
220 {
221 nodeX.parent = rootY;
222 }
223 else if (nodeX.rank > nodeY.rank)
224 {
225 nodeY.parent = rootX;
226 }
227 else
228 {
229 nodeY.parent = rootX;
230 ++nodeX.rank; // 秩相同，合并后根节点秩+1
231 }
232 }
233
240 bool connected(const T& x, const T& y) { return find(x) == find(y); }
241
246 size_t countSets() const
247 {
248 size_t count = 0;
249 for (const auto& pair : nodes)
250 {
251 // 根节点的父节点是自身
252 if (pair.first == pair.second.parent) ++count;
253 }
254 return count;
255 }
256
261 inline size_t size() const { return nodes.size(); }
262 };
