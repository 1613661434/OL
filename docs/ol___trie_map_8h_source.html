61 void traverse(std::shared_ptr<TrieNode<V>> node, std::string& path, std::list<std::string>& res)
62 {
63 if (!node) return;
64
65 if (node->isValid) // 用标记位判断是否为有效键
66 {
67 res.push_back(path);
68 }
69
70 for (auto& [c, child] : node->children)
71 {
72 path.push_back(c);
73 traverse(child, path, res);
74 path.pop_back();
75 }
76 }

86 void traverseByPattern(std::shared_ptr<TrieNode<V>> node, std::string& path,
87 const std::string& pattern, size_t i, std::list<std::string>& res)
88 {
89 if (!node) return; // 节点不存在，匹配失败
90
91 // 模式匹配完成
92 if (i == pattern.length())
93 {
94 // 如果当前节点为有效键
95 if (node->isValid) res.push_back(path);
96 return;
97 }
98
99 unsigned char c = pattern[i];
100 if (c == '.')
101 {
102 // 通配符匹配任意字符，尝试所有可能的子节点
103 for (auto& [ch, child] : node->children)
104 {
105 path.push_back(ch);
106 traverseByPattern(child, path, pattern, i + 1, res);
107 path.pop_back();
108 }
109 }
110 else
111 {
112 // 普通字符，查找对应的子节点
113 auto it = node->children.find(c);
114 if (it != node->children.end())
115 {
116 path.push_back(c);
117 traverseByPattern(it->second, path, pattern, i + 1, res);
118 path.pop_back();
119 }
120 }
121 }

131 std::shared_ptr<TrieNode<V>> put(std::shared_ptr<TrieNode<V>> node, const std::string& key, V val, size_t i)
132 {
133 if (!node) node = std::make_shared<TrieNode<V>>();
134
135 if (i == key.length())
136 {
137 node->val = val;
138 node->isValid = true; // 标记为有效节点（无论val是否为默认值）
139 return node;
140 }
141
142 unsigned char c = key[i];
143 node->children[c] = put(node->children[c], key, val, i + 1);
144 return node;
145 }

154 std::shared_ptr<TrieNode<V>> remove(std::shared_ptr<TrieNode<V>> node, const std::string& key, size_t i)
155 {
156 if (!node) return nullptr; // 节点不存在，直接返回
157
158 // 到达键的末尾，标记为无效
159 if (i == key.length())
160 {
161 node->isValid = false;
162 }
163 else
164 {
165 unsigned char c = key[i];
166 auto childIt = node->children.find(c);
167 if (childIt == node->children.end()) return node;
168
169 // 递归删除子节点；子树被完全剪枝时必须擦除映射，不能留下空shared_ptr。
170 auto child = remove(childIt->second, key, i + 1);
171 if (child)
172 childIt->second = std::move(child);
173 else
174 node->children.erase(childIt);
175 }
176
177 // 后序处理：如果节点有效（isValid=true），保留节点
178 if (node->isValid) return node;
179
180 // 如果没有子节点，删除当前节点
181 if (node->children.empty()) return nullptr;
182
183 // 有子节点但无值，保留节点作为中间路径
184 return node;
185 }

194 bool hasPattern(std::shared_ptr<TrieNode<V>> node, const std::string& pattern, size_t i)
195 {
196 if (!node) return false; // 节点不存在，匹配失败
197
198 // 模式处理完毕，检查当前节点是否有值
199 if (i == pattern.length()) return node->isValid;
200
201 unsigned char c = pattern[i];
202 if (c != '.')
203 {
204 // 普通字符，查找对应子节点并递归匹配
205 auto it = node->children.find(c);
206 return (it != node->children.end()) && hasPattern(it->second, pattern, i + 1);
207 }
208 else
209 {
210 // 通配符，尝试所有子节点
211 for (auto& [ch, child] : node->children)
212 {
213 if (hasPattern(child, pattern, i + 1)) return true;
214 }
215 return false;
216 }
217 }

225 std::shared_ptr<TrieNode<V>> findNode(std::shared_ptr<TrieNode<V>> node, const std::string& key)
226 {
227 auto current = node;
228 for (size_t i = 0; i < key.length(); ++i)
229 {
230 if (!current) return nullptr; // 路径中断，返回null
231
232 // 查找下一个字符对应的子节点
233 auto it = current->children.find(key[i]);
234 if (it == current->children.end()) return nullptr; // 子节点不存在，返回null
235 current = it->second;
236 }
237 return current; // 返回键末尾对应的节点
238 }

307 std::string shortestPrefix(const std::string& query)
308 {
309 auto current = root;
310 for (size_t i = 0; i < query.length(); ++i)
311 {
312 if (!current) break;
313
314 // 基于isValid判断，而非val是否为默认值
315 if (current->isValid)
316 {
317 return query.substr(0, i);
318 }
319
320 auto it = current->children.find(query[i]);
321 if (it == current->children.end()) break;
322 current = it->second;
323 }
324
325 if (current && current->isValid) return query;
326
327 return "";
328 }

335 std::string longestPrefix(const std::string& query)
336 {
337 auto current = root;
338 size_t max_len = 0; // 记录最长有效前缀的长度
339 size_t i; // 循环变量（在循环外定义，用于判断是否遍历完所有字符）
340
341 for (i = 0; i < query.length(); ++i)
342 {
343 if (!current) break; // 路径中断，退出循环
344
345 // 如果当前节点是有效键，更新最长前缀长度
346 if (current->isValid) max_len = i;
347
348 // 查找下一个字符对应的子节点
349 auto it = current->children.find(query[i]);
350 if (it == current->children.end()) break; // 字符不匹配，退出循环
351 current = it->second; // 移动到子节点
352 }
353
354 // 只有当：1. 遍历完查询字符串的所有字符；2. 最终节点是有效键
355 // 才返回整个查询字符串（说明查询字符串本身就是已插入的键）
356 if (i == query.length() && current && current->isValid) return query;
357
358 // 否则返回最长有效前缀
359 return query.substr(0, max_len);
360 }
