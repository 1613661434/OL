60 void init()
61 {
62 for (size_t i = 0, size = b.size(); i < size; ++i)
63 {
64 size_t j = i + lowbit(i + 1);
65 if (j < size) b[j] += b[i];

93 ++idx;

109 T rangeSum(size_t left, size_t right) const
110 {

112 if (right >= b.size()) right = b.size() - 1;
113 T leftSum = (left == 0) ? 0 : sum(left - 1);
114 return sum(right) - leftSum;
