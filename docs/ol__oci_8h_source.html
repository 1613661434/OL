2/*
3 * 程序名：ol_oci.h
4 * 功能描述：Oracle数据库操作实现，适配连接池，支持以下特性：
5 * - 连接管理：支持数据库连接、重连、断开、状态检测
6 * - 事务控制：支持手动事务提交/回滚、自动提交配置
7 * - 预处理语句：支持参数绑定、结果集绑定、SQL格式化预处理
8 * - 数据类型：支持int/long/float/double/string/BLOB/CLOB全类型绑定
9 * - 大字段操作：支持文件与BLOB/CLOB互转、分块传输大文件
10 * - 错误处理：统一执行结果码、错误信息、受影响行数返回
11 * 作者：ol
12 * 适用标准：C++17及以上
13 */

16/*
17// OCI 返回状态码
18OCI_SUCCESS 0 // maps to SQL_SUCCESS of SAG CLI 函数执行成功
19OCI_SUCCESS_WITH_INFO 1 // maps to SQL_SUCCESS_WITH_INFO 执行成功，但有诊断消息返回，
20 // 可能是警告信息，但是，在测试的时候，我还从未见
21 // 识到OCI_SUCCESS_WITH_INFO是怎么回事
22OCI_RESERVED_FOR_INT_USE 200 // reserved
23OCI_NO_DATA 100 // maps to SQL_NO_DATA 函数执行完成，但没有其他数据
24OCI_ERROR -1 // maps to SQL_ERROR 函数执行错误
25OCI_INVALID_HANDLE -2 // maps to SQL_INVALID_HANDLE 传递给函数的参数为无效句柄，
26 // 或传回的句柄无效
27OCI_NEED_DATA 99 // maps to SQL_NEED_DATA 需要应用程序提供运行时刻的数据
28OCI_STILL_EXECUTING -3123 // OCI would block error 服务环境建立在非阻塞模式，
29 // OCI函数调用正在执行中
30
31OCI_CONTINUE -24200 // Continue with the body of the OCI function
32 // 说明回调函数需要OCI库恢复其正常的处理操作
33OCI_ROWCBK_DONE -24201 // done with user row callback
34*/
