83 void lock() noexcept
84 {
85 const std::thread::id current_thread = std::this_thread::get_id();
86 const std::thread::id current_owner = owner_thread.load(std::memory_order_acquire);
87
88 // 同一线程：递归加锁，递增计数
89 if (current_thread == current_owner)
90 {
91 recursion_count.fetch_add(1, std::memory_order_relaxed);
92 return;
93 }
94
95 // 不同线程：自旋等待获取锁
96 while (flag.test_and_set(std::memory_order_acquire));
97
98 // 记录持有者和初始计数
99 owner_thread.store(current_thread, std::memory_order_release);
100 recursion_count.store(1, std::memory_order_relaxed);
101 }
102
107 bool try_lock() noexcept
108 {
109 const std::thread::id current_thread = std::this_thread::get_id();
110 const std::thread::id current_owner = owner_thread.load(std::memory_order_acquire);
111
112 // 同一线程：直接递增计数，返回成功
113 if (current_thread == current_owner)
114 {
115 recursion_count.fetch_add(1, std::memory_order_relaxed);
116 return true;
117 }
118
119 // 不同线程：尝试获取锁，无阻塞
120 if (!flag.test_and_set(std::memory_order_acquire))
121 {
122 owner_thread.store(current_thread, std::memory_order_release);
123 recursion_count.store(1, std::memory_order_relaxed);
124 return true;
125 }
126
127 // 锁已被其他线程持有，返回失败
128 return false;
129 }
130
134 void unlock()
135 {
136 const std::thread::id current_thread = std::this_thread::get_id();
137 const std::thread::id current_owner = owner_thread.load(std::memory_order_acquire);
138
139 // 非持有线程解锁：抛逻辑错误异常
140 if (current_thread != current_owner)
141 {
142 throw std::logic_error("recursive_spin_mutex::unlock(): called by non-owner thread");
143 }
144
145 // 递减计数，判断是否需要释放锁
146 const int new_count = recursion_count.fetch_sub(1, std::memory_order_relaxed) - 1;
147 if (new_count <= 0)
148 {
149 recursion_count.store(0, std::memory_order_relaxed);
150 owner_thread.store(std::thread::id(), std::memory_order_relaxed);
151 // owner和递归计数必须在发布空闲状态前清除，避免新持有者的数据被旧线程覆盖。
152 flag.clear(std::memory_order_release);
153 }
154 }
155 };
