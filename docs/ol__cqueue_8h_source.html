2/*
3 * 程序名：ol_cqueue.h
4 * 功能描述：固定大小的循环队列（circular queue）模板类，支持以下特性：
5 * - 基于静态数组实现，大小由模板参数指定
6 * - 支持POD类型和非POD类型，自动适配内存管理
7 * - 提供入队、出队、查看队头、判断空满等基础操作
8 * - 支持移动构造和移动赋值，禁用拷贝构造和赋值（避免资源冲突）
9 * - 支持原地构造元素（emplace）以提升性能
10 * 作者：ol
11 * 适用标准：C++17及以上
12 */

66 void moveFrom(cqueue&& other)
67 {
68 m_inited = other.m_inited;
69 m_size = 0;
70 m_front = 0;
71 m_rear = MAX_SIZE - 1;
72
73 if (!m_inited) return;
74
75 try
76 {
77 for (size_t i = 0; i < other.m_size; ++i)
78 {
79 const size_t src_idx = (other.m_front + i) % MAX_SIZE;
80 emplace(std::move(*other.elementAt(src_idx)));
81 }
82 }
83 catch (...)
84 {
85 clear();
86 throw;
87 }
88
89 other.clear();
90 }

136 m_size = 0; // 队列的实际长度。

138 // m_data是原始存储，元素在push/emplace时才开始生命周期。
139 }

145 inline bool full() const { return m_size == MAX_SIZE; }
146
151 inline bool empty() const { return m_size == 0; }

158 bool push(const T& e)
159 {
160 if (full())
161 {
162 std::cerr << "Circular queue is full, enqueue failed.\n";
163 return false;
164 }
165
166 const size_t new_rear = (m_rear + 1) % MAX_SIZE;
167 new (&m_data[new_rear]) T(e);
168 m_rear = new_rear;
169 ++m_size;
170
171 return true;
172 }

179 bool push(T&& e)
180 {
181 if (full())
182 {
183 std::cerr << "Circular queue is full, enqueue failed.\n";
184 return false;
185 }
186 const size_t new_rear = (m_rear + 1) % MAX_SIZE;
187 new (&m_data[new_rear]) T(std::move(e));
188 m_rear = new_rear;
189 ++m_size;
190 return true;
191 }
192
197 bool pop()
198 {
199 if (empty()) return false;
200
201 destroyAt(m_front);
202 m_front = (m_front + 1) % MAX_SIZE; // 队列头指针后移。
203 --m_size;
204
205 if (m_size == 0)
206 {
207 m_front = 0;
208 m_rear = MAX_SIZE - 1;
209 }
210
211 return true;
212 }

218 void clear()
219 {
220 if (!m_inited || empty()) return; // 未初始化或已空则直接返回
221
222 for (size_t i = 0; i < m_size; ++i)
223 {
224 const size_t index = (m_front + i) % MAX_SIZE;
225 destroyAt(index);
226 }
227
228 // 重置队列状态（无需清空数组内存，后续操作会覆盖）
229 m_front = 0;
230 m_rear = MAX_SIZE - 1;
231 m_size = 0;
232 }

268 template <typename... Args>
269 bool emplace(Args&&... args)
270 {
271 if (full())
272 {
273 std::cerr << "Circular queue is full, enqueue failed.\n";
274 return false;
275 }
276 const size_t new_rear = (m_rear + 1) % MAX_SIZE;
277 new (&m_data[new_rear]) T(std::forward<Args>(args)...);
278 m_rear = new_rear;
279 ++m_size;
280 return true;
281 }

287 void print() const

292 std::cout << "m_data[" << index << "],value=" << *elementAt(index) << '\n';
293 }
294 }
